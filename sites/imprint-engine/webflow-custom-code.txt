<script>
  // English fallback until a native Webflow locale is configured.
  if (!document.documentElement.lang) document.documentElement.lang = "en";
  // Resolve main once. CSS and JS always come from the same commit.
  const mhRepo = "MikeRudi/mharris-webflow";
  const mhSource = new Map();
  window.mhVersion = fetch(`https://api.github.com/repos/${mhRepo}/commits/main?v=${Date.now()}`, { cache: "no-store" })
    .then((response) => { if (!response.ok) throw new Error("Commit lookup failed"); return response.json(); })
    .then((commit) => commit.sha).catch(() => "main");
  window.mhFetch = function (path) {
    if (!mhSource.has(path)) mhSource.set(path, mhVersion.then(async (version) => {
      const response = await fetch(`https://raw.githubusercontent.com/${mhRepo}/${version}/${path}`,
        { cache: version === "main" ? "no-store" : "force-cache" });
      if (!response.ok) throw new Error("Could not load: " + path);
      return response.text();
    }));
    return mhSource.get(path);
  };
  window.mhLoad = async function (path, type) {
    const element = document.createElement(type === "css" ? "style" : "script");
    element.textContent = await mhFetch(path);
    document[type === "css" ? "head" : "body"].appendChild(element);
  };
  mhLoad("sites/imprint-engine/src/styles.css", "css").catch(console.error);
  // Fetch early; execute only after deferred libraries and Webflow's jQuery are ready.
  mhFetch("sites/imprint-engine/src/script.js").catch(console.error);
</script>
